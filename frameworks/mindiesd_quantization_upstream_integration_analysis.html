01 / 先说结论
What to change

把 MindIE layer 改成可被调用的运行时对象

不是让 MindIE 接管模型，也不是设计一套新格式。只增加一个很薄的 Python 接口：上游加载好权重后调用 prepare()，推理时调用 forward()。

最终调用关系只有一条：上游 Linear / Attention → 框架自己的量化方法 → MindIE runtime layer。

SGLang 和 vLLM-Omni 继续负责模型、权重加载、并行和调度。MindIE 只负责量化权重准备、激活量化和 NPU 算子执行。

02 / 先改 MindIE
Patch 1 · accelerator library

增加 runtime facade，原 layer 算子代码继续用

Facade 就是“很薄的统一入口”。它不懂 SGLang、vLLM、模型名字和文件路径，只接收 tensor、shape 与当前 step。

MindIE-SD/mindiesd/runtime_quant/ ├── api.py # 公共输入/输出类型 ├── linear.py # 包装现有量化 Linear ├── attention.py # 包装现有量化 FA └── registry.py # algorithm → runtime class 继续复用 mindiesd/quantization/layer.py mindiesd/layers/quant/* torch_npu / torch.ops.mindiesd.*

现有代码真正的问题

例如 W8A8QuantLinear.__init__ 通过 weights + prefix 自己取 tensor。上游框架已经加载并分片了权重，不能再让 MindIE 从文件取一次。

  • 保留 quant_matmul()。
  • 把 _init_*_quant_param() 拆成接收 tensor 的 prepare()。
  • 旧构造器调用新 prepare()，继续兼容。
mindiesd/runtime_quant/api.py建议新增
from dataclasses import dataclass
from typing import Mapping, Protocol
import torch

@dataclass(frozen=True)
class RuntimeContext:
    step_index: int | None = None

@dataclass(frozen=True)
class LinearMeta:
    algorithm: str          # "w8a8", "mxfp8", "mxfp4" ...
    input_size: int
    output_size: int
    dtype: torch.dtype
    use_nz: bool = False

class QuantLinearRuntime(Protocol):
    def prepare(
        self, tensors: Mapping[str, torch.Tensor], meta: LinearMeta
    ) -> object: ...

    def forward(
        self, x: torch.Tensor, state: object,
        bias: torch.Tensor | None = None,
        context: RuntimeContext | None = None,
    ) -> torch.Tensor: ...
mindiesd/runtime_quant/linear.pyW8A8 示例
@dataclass(frozen=True)
class W8A8State:
    weight: torch.Tensor
    weight_scale: torch.Tensor

class W8A8Runtime:
    def prepare(self, tensors, meta):
        # 从原 W8A8QuantLinear._init_dynamic_quant_param() 提取
        weight = tensors["weight"]
        weight_scale = tensors["weight_scale"].squeeze().to(meta.dtype)
        if meta.use_nz:
            weight = torch_npu.npu_format_cast(weight.npu(), 29).T
        else:
            weight = weight.T
        return W8A8State(weight=weight, weight_scale=weight_scale)

    def forward(self, x, state, bias=None, context=None):
        # 直接复用原 W8A8QuantLinear.quant_matmul() 主体
        original_shape = x.shape
        x = x.reshape(-1, x.shape[-1])
        qx, x_scale = torch_npu.npu_dynamic_quant(x)
        out = torch_npu.npu_quant_matmul(
            qx, state.weight, state.weight_scale,
            pertoken_scale=x_scale,
            bias=bias,
            output_dtype=x.dtype,
        )
        return out.reshape(*original_shape[:-1], -1)

def create_linear_runtime(algorithm: str) -> QuantLinearRuntime:
    return {
        "w8a8": W8A8Runtime,
        "mxfp8": MXFP8Runtime,
        "mxfp4": MXFP4Runtime,
        "mxfp4_dualscale": MXFP4DualScaleRuntime,
    }[algorithm]()

class W8A8QuantLinear(nn.Module):
    def __init__(
        self, in_features, out_features,
        bias=True, weights=None, prefix=None, **kwargs
    ):
        super().__init__()
        tensors = {
            "weight": get_quant_weight(weights, f"{prefix}.weight"),
            "weight_scale": get_quant_weight(weights, f"{prefix}.weight_scale"),
        }
        loaded_bias = (
            get_quant_weight(weights, f"{prefix}.bias") if bias else None
        )
        self.register_buffer("bias", loaded_bias, persistent=False)
        self.runtime = W8A8Runtime()
        self.state = self.runtime.prepare(
            tensors,
            LinearMeta(
                algorithm="w8a8",
                input_size=in_features,
                output_size=out_features,
                dtype=kwargs.get("dtype", torch.bfloat16),
                use_nz=kwargs.get("use_nz", False),
            ),
        )

    def forward(self, x):
        return self.runtime.forward(x, self.state, self.bias)

import torch

from mindiesd.runtime_quant import (
    LinearMeta, RuntimeContext, create_linear_runtime
)
from sglang.multimodal_gen.runtime.layers.linear import LinearBase, LinearMethodBase
from sglang.multimodal_gen.runtime.layers.quantization import register_quantization_config
from sglang.multimodal_gen.runtime.layers.quantization.configs.base_config import QuantizationConfig
from sglang.multimodal_gen.runtime.managers.forward_context import get_forward_context

@register_quantization_config("mindie_runtime")
class MindIERuntimeConfig(QuantizationConfig):
    def __init__(self, algorithm="mxfp8", use_nz=False):
        super().__init__()
        self.algorithm = algorithm
        self.use_nz = use_nz

    @classmethod
    def get_name(cls): return "mindie_runtime"

    @classmethod
    def get_supported_act_dtypes(cls):
        return [torch.bfloat16, torch.float16]

    @classmethod
    def get_min_capability(cls): return 0  # 不是 CUDA capability

    @staticmethod
    def get_config_filenames(): return []

    @classmethod
    def from_config(cls, config):
        return cls(
            algorithm=config.get("algorithm", "mxfp8"),
            use_nz=config.get("use_nz", False),
        )

    def get_quant_method(self, layer, prefix):
        if isinstance(layer, LinearBase):
            return MindIERuntimeLinearMethod(self)
        return None

class MindIERuntimeLinearMethod(LinearMethodBase):
    def __init__(self, config):
        self.config = config
        self.runtime = create_linear_runtime(config.algorithm)

    def create_weights(self, layer, *args, **kwargs):
        # 沿用 SGLang 现有 ModelWeightParameter + weight_loader 写法。
        # 这里注册 weight / weight_scale / bias 等参数，不让 MindIE 读文件。
        register_sglang_quant_parameters(layer, self.config, *args, **kwargs)

    def process_weights_after_loading(self, layer):
        tensors = collect_quant_tensors(layer, self.config.algorithm)
        layer._mindie_state = self.runtime.prepare(
            tensors,
            LinearMeta(
                algorithm=self.config.algorithm,
                input_size=layer.input_size_per_partition,
                output_size=layer.output_size_per_partition,
                dtype=layer.orig_dtype,
                use_nz=self.config.use_nz,
            ),
        )

    def apply(self, layer, x, bias=None):
        ctx = get_forward_context()
        return self.runtime.forward(
            x, layer._mindie_state, bias,
            RuntimeContext(step_index=ctx.current_timestep),
        )

# 上面两个 helper 的实现不是新框架：
# 1. register_sglang_quant_parameters：直接保留现有
#    mxfp8_npu.py::create_weights() 的 ModelWeightParameter 注册代码。
# 2. collect_quant_tensors：只做下面这种属性收集。
def collect_quant_tensors(layer, algorithm):
    names = REQUIRED_TENSORS[algorithm]
    return {name: getattr(layer, name).data for name in names}

        self.algorithm = algorithm
        self.use_nz = use_nz

    @classmethod
    def get_name(cls): return "mindie_runtime"

    @classmethod

class MindIERuntimeLinearMethod(LinearMethodBase):
    def __init__(self, config):
        self.config = config
        self.runtime = create_linear_runtime(config.algorithm)

    def create_weights(self, layer, *args, **kwargs):
        # 复用 vLLM 的参数类和 weight_loader；只注册所需 tensor。
        register_vllm_quant_parameters(layer, self.config, *args, **kwargs)

    def process_weights_after_loading(self, layer):
        layer._mindie_state = self.runtime.prepare(
            collect_quant_tensors(layer, self.config.algorithm),
            build_linear_meta_from_vllm(layer, self.config),
        )

    def apply(self, layer, x, bias=None):
        step = None
        if is_forward_context_available():
            step = get_forward_context().denoise_step_idx
        return self.runtime.forward(
            x, layer._mindie_state, bias,
            RuntimeContext(step_index=step),
        )

06 / 为什么这样改
Relevant analysis only

两个多模态框架的接入点其实高度一致

只保留与改代码直接相关的差异。主仓量化设计不再单列长篇分析。

问题SGLang multimodal_genvLLM-OmniMindIE 应对
Linear 挂载点LinearMethodBasevLLM LinearMethodBase同一 runtime 被两个薄 method 调用
加载后处理process_weights_after_loadingprocess_weights_after_loading都在这里调用 prepare()
多模态作用域transformer loader/config 传递ComponentQuantizationConfig 最长前缀路由MindIE 不参与组件选择
当前 stepforward_context.current_timestepForwardContext.denoise_step_idx转换成 RuntimeContext.step_index
Attention独立 AttentionBackend + NPU resolver独立 Diffusion backend registry分别写薄 backend，复用同一 FA runtime
多平台已有 CUDA/ROCm/NPU/XPU platform已有 CUDA/ROCm/NPU/XPU/MUSA platformMindIE 只注册 NPU；其他平台保持原实现
为什么还提公共接口：这里只是为了不把 torch_npu 写进 SGLang/vLLM 的公共量化基类。它不是一套大型 ABI 工程。Python 层只需要稳定的 prepare/forward；GPU 等其他加速库可以提供同名接口，也可以继续使用框架现有实现。
07 / 怎么落地
PR order & tests

按四个小 PR 提交，不做大重构

先跑通 W8A8 或 MXFP8 Linear，再接 MXFP4 和 FA。每一步都能和当前上游 NPU 实现做结果对比。

MindIE runtime facade

新增 tensor 输入接口；旧 layer 改为内部调用 facade。单测确认改造前后同输入完全一致。

SGLang multimodal_gen bridge

只接一个算法、一个 DiT 模型；验证加载、TP=1/2、CPU offload 与多 step。

vLLM-Omni bridge

复用 component config，仅量化 transformer；与现有 NPU method 做逐层输出和整段 latent trajectory 对比。

FA + 其他算法

接量化 AttentionBackend，再迁移 MXFP4、DualScale、MoE。旧重复实现稳定两个版本后再删除。

Import

安装 MindIE runtime wheel 后，两个 bridge 可独立 import；不拉入 graph rewrite 模块。

Numerics

同一 tensor 对比旧 MindIE layer、runtime facade 和两个上游 bridge。

Parallel

Row/Column/QKV Linear 在分片后 prepare，不能再次切分或转置错误。

Concurrency

交错执行不同 step，请求结果必须等于逐请求串行结果。

首个可合入版本：建议选择 W8A8 dynamic 或 MXFP8 online Linear。它们没有复杂逐层静态 scale/offset，最容易证明“上游加载权重，MindIE 只负责运行”的边界。
